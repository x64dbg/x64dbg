; A normal software breakpoint must interrupt tracing and clear the fast-run state.
settingset Events, SystemBreakpoint, 1
settingset Events, EntryBreakpoint, 0
init "@TARGET@"
bp trace_party:Ready
run
bpc trace_party:Ready
bp trace_party:TraceBegin
run
bpc trace_party:TraceBegin
bp trace_party:UserCallback
TraceSetStepFilter user, run
TraceIntoConditional "cip==trace_party:TraceEnd", .50000
testassert cip == trace_party:UserCallback, "ordinary breakpoint interrupted the trace"
testassert dword:[trace_party:gCallbacks] == 0, "first callback has not executed yet"
bpc trace_party:UserCallback
bp trace_party:Finished
TraceSetStepFilter user, run
TraceIntoConditional "cip==trace_party:TraceEnd", .50000
testassert cip == trace_party:TraceEnd, "second trace completed after breakpoint cleanup"
testassert dword:[trace_party:gCallbacks] == 3, "callbacks were not skipped or duplicated"
run
testassert cip == trace_party:Finished, "normal execution resumed"
testassert dword:[trace_party:gProtectionFailures] == 0, "breakpoint cleanup restored page protections"
bpc trace_party:Finished
run
