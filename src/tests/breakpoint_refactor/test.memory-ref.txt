settingset Events, EntryBreakpoint, 0
init tests/breakpoint_refactor.exe
alloc 1000
$page = $result
testassert $page != 0, "allocated a dedicated memory-breakpoint page"
bpmrange $page, 1000, w
bprstate mem, $page, 1, 1, ref
bpmd $page
bprstate mem, $page, 0, 0, ref
bpme $page
bprstate mem, $page, 1, 1, ref
bpmc $page
free $page
testassert $result == 1, "freed the memory-breakpoint page"
